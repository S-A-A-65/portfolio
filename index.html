public async Task<ServiceResult<AppointmentDto>> BookAppointmentAsync(CreateAppointmentDto dto)
{
    // Prevent double booking with LINQ query optimization
    bool isConflict = await _context.Appointments
        .AnyAsync(a => a.DoctorId == dto.DoctorId 
                    && a.AppointmentDate == dto.AppointmentDate 
                    && a.Status != AppointmentStatus.Cancelled);

    if (isConflict)
        return ServiceResult<AppointmentDto>.Failure("Doctor is already booked at this time slot.");

    var entity = _mapper.Map<Appointment>(dto);
    _context.Appointments.Add(entity);
    await _context.SaveChangesAsync();

    return ServiceResult<AppointmentDto>.Success(_mapper.Map<AppointmentDto>(entity));
}

Optimized T-SQL Stored Procedure Snippet

sp_GetOptimizedPatientAnalytics.sql T-SQL / SQL Server 
CREATE PROCEDURE dbo.sp_GetOptimizedPatientAnalytics
    @StartDate DATETIME2,
    @EndDate DATETIME2
AS
BEGIN
    SET NOCOUNT ON;
    
    SELECT 
        d.DoctorId,
        d.FullName AS DoctorName,
        COUNT(a.AppointmentId) AS TotalAppointments,
        SUM(CASE WHEN a.Status = 'Completed' THEN 1 ELSE 0 END) AS CompletedCount
    FROM dbo.Doctors d WITH (NOLOCK)
    INNER JOIN dbo.Appointments a WITH (NOLOCK) ON d.DoctorId = a.DoctorId
    WHERE a.AppointmentDate BETWEEN @StartDate AND @EndDate
    GROUP BY d.DoctorId, d.FullName
    ORDER BY TotalAppointments DESC;
END;
